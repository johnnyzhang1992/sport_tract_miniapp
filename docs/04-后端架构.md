```js
    highAccuracy: true,       // 高精度定位开关
  },
  createdAt: Date,
  updatedAt: Date,
}
```

### Activity
```js
{
  _id: ObjectId,
  userId: ObjectId,                       // 索引（查询用户活动）
  type: String,                           // hiking|walking|running|cycling|mountaineering|swimming
  status: 'in_progress' | 'finished',
  startTime: Number,
  endTime: Number,
  duration: Number,                       // 秒（运动时长 = 墙钟 − pausedMs − standstillMs）
  standstillMs: Number,                   // 自动暂停：finish 时判出的静止时段总时长（毫秒）
  distance: Number,                       // 米
  avgPace: Number,                        // 秒/公里
  calories: Number,
  elevationGain: Number,                  // 米（滤波后）
  maxAltitude: Number,
  startAddress: String,
  endAddress: String,
  trackPoints: [{
    seq: Number,                          // 客户端序号，幂等去重键
    lat: Number, lng: Number,
    altitude: Number | null,
    speed: Number,
    pauseGap: Boolean,                    // 暂停恢复后首个有效点（渲染断开连线）
    still: Boolean,                       // finish 时判出的静止时段点（自动暂停口径，见 §5 静止剔除）
    timestamp: Number,
  }],                                     // 索引：{ activityId: 1, 'trackPoints.seq': 1 }（复合唯一，可选）
  markers: [{
    id: String,                           // 客户端生成 ID
    lat: Number, lng: Number,
    timestamp: Number,
    type: 'checkpoint'|'rest'|'photo'|'note',
    note: String,
    photoUrl: String,
    address: String,
  }],
  lastPointSeq: Number,                   // 已接收最大 seq
  pausedMs: Number,
  deviceInfo: Object,
  createdAt: Date,
  updatedAt: Date,
}
```

**性能考虑**：长轨迹点较多，按 `seq` 追加用 `$push` + `$max lastPointSeq`；轨迹点数组上限（如 2 万点）保护，超出时服务端提示客户端抽稀。

**导入/导出口径（2026-09-24 核对）**
- **导入也做静止剔除**：`importActivity`（`POST /api/activities/import`，multipart）原先只写 `duration = 末点 − 首点`（墙钟）、不写 `pausedMs/standstillMs`，于是「同一条徒步：导进来 = 墙钟、App 里录 = 净时长」两条路径对不上。现已与 `finish` 共用 `markStandstill`：静止时段照剔、写 `standstillMs`、点打 `still`、`duration` 用净时长（导入文件没有手动暂停语义，`pausedMs` 恒 0）。注意导入的「<1m 相邻重复点合并」会压掉**完全同坐标**的静止段，那类文件不剔该段（合成数据测试时要带抖动，否则测不到）。
- **坐标协议（2026-09-24 已修）**：录制落库是 **GCJ-02**（`record` 用 `type: 'gcj02'`），而 GPX/GeoJSON 等标准协议是 **WGS-84**——`toGpx` 原先原样写出库内坐标，后果有两个：(a) 外部工具（Strava / Google Earth / 两步路）打开我们的导出偏数百米；(b) **导出再导入会二次偏移**（`importActivity` 按 WGS-84 又转一次），用户实测 588.7m、真机复现。现导出侧对 trkpt 与 wpt 都做 `gcj02ToWgs84`（`utils/coordinate.ts` 新增逆变换，不动点迭代 3 次，境外原样返回），往返闭合（真实活动 1086 点往返误差 0.000m；回归用例 `test/activity.test.ts`「导出 GPX → 再导入」+ `test/coordinate.test.ts` 逆变换往返 <1m）。**注意：修之前导出的 .gpx 里是 GCJ-02，不要再导入**（会偏）。
- 解析器只取 lat/lng/ele/time，`speed` 硬编码为 `null`（`import.ts` 落库处）——来源文件里的 speed/扩展字段目前被丢弃。
- **管理端导出（2026-09-24）**：`GET /api/admin/activities/:id/gpx`（adminAuth，按 id 直取、不校验归属）复用同一份 `toGpx`，用**全量轨迹点**（详情接口为弹窗图表抽稀到 600 点，导出不抽）。**导出不删点**：车速段的点照导，`vehicle` 剔除只作用在入库指标上——删点会让「导出→再导入」真的丢数据，而导入端本来就会重跑一遍 `markVehicle`，同一份坐标得到同一判定。也不往 GPX 写 `<extensions>` 标 vehicle/still（标准工具不读、我们的导入器也不看）。回归：`test/admin-detail.test.ts`「管理员导出 GPX」（trkpt 数 = 库内点数、坐标与 `gcj02ToWgs84` 一致且与库内原值差 >300m，漏转必红；400/404/401 各一条）。**后台按钮已浏览器实测（2026-09-24）**：dev hiking `6a9ed90e…` 点「导出 GPX」→ `GET /api/admin/activities/<id>/gpx` 200 `application/gpx+xml`，落盘 `activity-6a9ed90e3bc4f491e962629c.gpx`（386KB）；文件里 **3120 个 trkpt == 库内 3120 点**（详情接口只给 600 点，证明没抽稀），每点最大坐标偏差 **0.0000m**、与库内 GCJ-02 原值相差 **589.3m**（即坐标系协议里那个「漏转就平移 580–590m」的量级）。

## 5. 同步协议细节（核心）

**目标**：运动中定时增量上传 + 断网本地暂存补传（决策 D13），服务端幂等，最终一致。

```
客户端：
  1. 采点 → 本地缓冲（带 seq，全局自增）
  2. 每 30s（或缓冲 > N 条）：POST /points {fromSeq, points[]}
  3. 失败（网络）：入本地待同步队列（持久化），恢复后按序重试
  4. 结束：POST /finish 提交 final 包（完整点集 + 指标）
服务端：
  1. 接收 points：过滤 seq <= lastPointSeq 的重复点，$push 新点，$set lastPointSeq
  2. finish：以客户端 final 包为准 → 全量替换 trackPoints + 重算指标（防漏点/脏点）
  3. 对账响应：{ status, lastPointSeq, activity }
```

**冲突处理**：同一活动单设备记录，无多端并发写；`finish` 后禁止再传 points（返回 409）。

### 5.1 静止剔除（自动暂停，2026-09-24）

用户在原地不动但**没点暂停**的时间，原先会被计进运动时长（`pausedMs` 只记手动暂停与「切后台无后台定位权限」「运动中退出页面」两种系统暂停）。修法放在服务端：`finish` 与 24h 自动收尾共用同一条管线，算一次入库，读接口不再重算。

- 算法：`src/utils/standstill.ts#markStandstill` —— 停留点（stay point）检测。以某点为锚向后扩展，只要后续点仍在锚点 **10m 三维半径**（水平 haversine + 高程差，见 `spatialDistance`）内就继续（不跨 `pauseGap`、不跨 >120s 断档）；扩展时长 ≥ **60s** 且点数 ≥ **3** 才算一段静止，给这些点打 `still`。
- 为什么用半径而不用速度：本项目点距 2–3 秒，静止时 GPS 抖动的瞬时速度与慢走同量级（都约 1–2 m/s），速度阈值分不开；「原地抖动走不出 10m、正常行进一定走得出」是几何事实。实测速率判据也不可用：速率 ≤0.5 m/s 几乎不改善（9.31km 徒步仍扣 3164s/60 段），≤0.3 m/s 又会把那条设备自报 speed=0 的 35s 真停丢掉。
- **判据是三维的（2026-09-24 追加，用户指出「爬坡或下坡坐标变化都可能很小」）**：之字上升 / 横向切坡时水平坐标可以长时间不出圈，只看二维会把「正在爬」判成休息。加高程后「水平不动但海拔在变」= 在爬、不算静止；海拔缺失/为 null 时按对退化（老数据不受影响）。**实测代价极小**：9.31km 徒步 1105s → 1102s、12.55km 634s → 599s，跑步/散步/骑行全为 0。注意 GPS 高程噪声本身就是 ±5m 量级，想在 60s 内增收 8m 需要 ~45° 坡，**这条是防御性的，别指望它显著改变数值**。
- 为什么 ≥3 个点：只有 2 个点、相隔几十秒落在同一圈内，说明是**没采到点**而不是静止，凭它删时间等于猜（实测 899 点那条轨迹里 16 段「停留」有 14 段只有 2 个点，加上这条门槛挡掉 626s）。
- 门槛/半径的标定（**2026-09-24 由 30s/15m 收紧为 60s/10m**）：先用 dev 库 9.31km 徒步量化「偏大」——63 段/3281s 占墙钟 29%、平均每 3 分钟一段，且每段段内**路径总长仅 13–23m**（与净位移几乎相等，排除「折返爬升被误判」）；同时段另外两条导入的 9.5km 徒步只扣 345s。根因是手机 GPS 只有坐标没有加速度计，「站住不动在漂」与「0.2 m/s 缓挪」的点序列同构，只能用「多长的静止才算休息」表达口径。实测档位（同一份数据）：60s→1528s/14 段、120s→835s/4 段；半径 10m 叠加后 **60s+10m → 1105s/8 段**，且跑步/散步的短停全部归零。门槛下限受几何约束：门槛 × 最慢真实速度 必须大于半径，否则匀速跑动必然被误伤（半径/门槛 = 0.167 m/s）。
- **断档期仍算静止（2026-09-24 已定夺，曾考虑「只扣采样连续的时间」）**：9.31km 徒步有 16 处 >30s 采样空档，其中 7 段（含最长 5.8 分钟那段）内部带 36–154s 空档、合计 1013s。否决改法的依据：这些空档两端的位移只有 **1.0–1.4m**（另两处 16.5m / 82.2m），坐标证据支持「人确实没挪窝」；改成只扣采样连续的时间会掉到 gap≤45s→370s、gap≤60s→525s，等于把真实休息还回去。
- 坡道上的实测反驳（同一轮）：按 ±150m 距离窗口算地形坡度，8 段里只有 2 段（合计 303s）位于真坡（+8.5% / +9.9%），而这两段的段内净海拔只有 +2.5m / −0.2m —— 说明是**停在坡上休息**，不是在缓爬。所以「坡度豁免」这个想法在本数据上站不住，别按它改。
- 对照组口径（有加速度计，不能直接抄）：华为「静止超过 10 秒自动暂停」、Keep 建议 15–30s、Strava 不公开阈值。
- 外部基准（2026-09-24，用户提供）：running `6ab40028…`（4.51km、无手动暂停、点间隔中位数 2s、最大空档 9s）——Apple Watch 训练时长 **36:44**；60s/10m 口径下我们 **36:47**（差 3s，属起止边界量级），而 30s/15m 口径下是 **35:20**（比手表少 84s）。这是「门槛过松」最硬的一条独立证据。（手表是否开了自动暂停未知：若开了，说明它认为这趟没有停顿；若没开，则 3s 的吻合只说明起止判定一致。）同一天的 9.31km 徒步手表给 2:58:55，而我们是 总 3:06:40 / 运动 2:48:18 —— 差 7:45 / 10:37，**两设备在跑步上口径一致、徒步上差这么多，更像手表侧的差异（未查证：是否有自动暂停、起止时刻是否对齐）**，不足以推翻坐标证据。
- 派生口径：净运动时长 = `endTime − startTime − pausedMs − standstillMs − vehicleMs`（下限 0），用它喂 `calcStats` → **平均配速与卡路里随之一并修正**；`totalDuration` 仍是墙钟，两者差值语义为「暂停 + 静止 + 车速段」。
- 已知边界（刻意接受）：净速度低于 `半径/门槛 ≈ 0.167 m/s` 的极慢移动（很陡的之字爬升）会被判为静止——三维判据能拦下「水平不动但持续升降」的那部分，拦不住在 10m 圈内水平缓挪又不升降的；抖动漂移 >10m 的手机可能漏判（宁可少扣）。
- 历史数据：`scripts/backfill-standstill.ts` 一次性回填（默认干跑，`--apply` 落库，闸门写成可重跑的不变量 `duration + standstillMs ≈ 墙钟`）。只改 `trackPoints.still / standstillMs / duration / avgPace`；`fastestKm` 不动（实测库里若干条与「用同一条轨迹重算」对不上，属 GPX 导入另一条路径的既有差异）。
- **分段用时的静止口径已改（2026-09-24，用户拍板）**：`calcFastestKm` 原先把 `still` 段的时间从 1km 窗口里剔掉，实测让线上那条记录从 3'41" 变成 **2'32"** —— 在 km 中间停 48 秒反而被奖励配速。现按**经过时间**算（与 Keep/华为的分段一致）：停下休息就是让这一公里变慢。副作用要知道：头部平均配速剔静止、分段不剔，所以同一条记录可能出现「最快 1km 比平均配速还慢」（线上那条就是 7'17" vs 6'42"），详情页单段明细加了一行灰字讲这件事。

### 5.2 非运动段剔除（疑似乘车，2026-09-24）

跑步/徒步途中搭车、推车、坐摆渡车的那一段，既不是自己动的、又会把 GPS 噪声「运」出真实位移，**位移与时长要一起剔**（与 §5.1 只剔时长不同）。动机是线上真实数据：`6ab3d14d…` 一条 5.95km「跑步」里 1.24km 是 25 km/h 移动出来的，刷出 `fastestKm 220.6`（3'41"）并占了全国榜「最快 1km」第一——算术没错，语义错了。

- 算法：`src/utils/vehicle.ts#markVehicle` —— 逐步速度 ≥ **6.5 m/s**（23.4 km/h，城市车流速度）、连续 ≥ **60s**、≥ **5 次定位**，段内允许穿插 ≤ **2 个慢步**（等红灯），遇 `pauseGap` 或 >120s 断档硬拆。命中段内的点打 `vehicle`，汇总写 `vehicleMs / vehicleM`。
- **为什么不是「高于人类极限」**：7.0 m/s（原方案）高于男子 1km 世界纪录均速 6.71 m/s 却在这条真实记录上判出 **0 段**——那 25 km/h 的逐步速度在 6.0–7.5 之间摆动，严格 ≥7.0 只剩 34s 碎段。所以阈值贴着车流速度，误伤风险改由「时长 + 步数 + 允许穿插」三条一起兜（单靠速度阈值兜不住：6.5 m/s 低于 1km 世界纪录均速 7.63 m/s）。
- 只对人力运动类型判：`running / walking / hiking / mountaineering`。骑行 10 m/s（36 km/h）、滑雪、划船都是正常速度，套这条会把自己判没。
- 管线位置：`cleanAltitudeSpikes → cleanTrajectory → smoothTrackSmart → markVehicle → markStandstill → calcStats/calcFastestKm`。车速段与静止段判据互斥（车速段不可能落在 10m 停留圈内），链式跑一次打标。导入路径（`services/import.ts`）同样接这两个，与 `finish`、24h 自动收尾三处同口径。
- **说明文案由服务端整句下发**：活动 DTO 带 `vehicleNotice`（`utils/vehicle.ts#formatVehicleNotice`），内容形如「约 1.26 公里疑似搭车，未计入」，无车速段为空串。挂在 `toActivityDto` 上，所以用户端 `/activities/:id`、列表与管理端 `/admin/activities/:id` 是同一句。只报距离（`x.xx 公里`，两位小数）：时长不进口号，它已经体现在「运动时长」比墙钟短这件事上，一句里摆两个数反而要解释谁是谁；段数也不报（用户不关心分了几段），但仍是「到底有没有车速段」的判据，从点标记现数（`markVehicle` 的 spans 只在判定时有值，库里不存）。单位固定公制（设置项里的「英制」目前没有任何渲染消费它）。**为什么放服务端**：这句原先在小程序详情页现拼，后台要显示同一句只能再抄一遍，两边迟早分叉；后台详情还把轨迹点抽稀到 ≤600 才下发，本地数出来的点数根本不是全量。回归 `test/vehicle-notice.test.ts`（两段/空串/长短距离取两位/脏数据 4 例，并钉住两个读接口一字不差）。
- 纪录闸门：`MIN_PLAUSIBLE_PACE_SEC_PER_KM = 130`（男子 1000m 世界纪录 2:11.05 ≈ 131 s/km），全国榜与个人最佳都不收这个值以内的数——回填之外的历史脏数据也一并挡住。
- 已知边界（刻意接受）：慢速搭车（<6.5 m/s，如 20 km/h 电动车）与「搭车 <60s」漏判——宁可少剔，也不把人真跑出来的数字删掉。
- 历史数据：**逻辑只有一份**，在 `src/services/vehicle-backfill.ts#backfillVehicle({apply,id,limit})`，两个入口共用——CLI `scripts/backfill-vehicle.ts`（薄壳：连库 + 把报告打成文本）与管理端 `POST /api/admin/activities/backfill-vehicle`（body `{apply?,id?,limit?}`）。**线上走接口**：运行镜像是 `npm ci --omit=dev`（没有 tsx）、`scripts/` 也不在镜像里。接口与 `recompute-elevation` 相反，**默认只算不写**、必须显式 `apply:true` 才落库（它会重写整条 trackPoints，误触代价不是一次重算）；报告首字段回 `db`（库名/主机）用来确认打到了哪个库，各清单截 50 条并另给 `*Count` 总数。可重跑：不变量闸门写成 `duration + standstillMs + vehicleMs ≈ 墙钟`；距离只在「库里 distance 与重算几何对得上」时改；`fastestKm` **只降不升**（回填不许顺手把人刷到更好名次，库里没值也不新造）。距离与配速直接复用 `calcStats` 而不是在脚本里重算，否则复跑会抖出 ±1s 的假差异。**dev 库实测（重构后复跑校验）**：`ts` 版 CLI 与容器打包版（esbuild 单文件）报告逐字一致，165 条待处理里 43 条判出车速段、0 条待更新（已回填过）；线上那条 → 距离 5951→4666m、运动时长 2411→1877s、`fastestKm` 3'41"→7'17"（该记录同时补 361s 静止）。取证造数：`scripts/seed-vehicle-test.ts --uid=<模拟器账号>` 再跑回填。

### 5.3 卡路里的体重口径（2026-09-24）

`calories = MET × 体重 × 净时长(h)`，MET 由运动类型给（`config/constants.ts#ACTIVITY_TYPE_META`），**体重只认服务端档案 `user.weightKg`**（`src/services/weight.ts#resolveWeightKg`，查不到用户才落 `DEFAULT_WEIGHT_KG = 60`）。

- 修的是：会落卡路里的 5 条路径（`finishActivity` / 24h 自动收尾 / 纠偏 `reprocessActivity` / 改类型 `updateActivityMeta` / GPX 导入）原先**只有 finish 传了体重，且传的是客户端 final 包上报值**，其余 4 条悄悄落到 60kg。于是同一用户「App 里录」和「第三方导入」两条记录卡路里能差 1.5 倍，改完档案再点纠偏还会把数字拽回 60kg。
- `FinishActivitySchema` 删掉了 `weightKg` 字段（zod 默认剥未知键，老客户端继续上传也不会 400，只是不再参与计算）；小程序 `tracker.js#buildFinalPack` 同步不再带这个键。**端上实时面板仍用自己的档案体重估算**（`services/tracker.js`），所以运动中显示的数字与结束后入库的数字同源。
- 已知边界（**存量未回填，待拍板**）：卡路里是入库时算死的，改口径不会自动追平历史行；§5.1/§5.2 的两个回填脚本缩短 `duration` 时也**刻意不碰 `calories`**。dev 库按新口径整体重算与库里值对比（1184 条 finished）：**300 条不一致**，其中 262 条只差 ±1–2 kcal（`duration` 入库前取整、重算拿取整值再乘，属四舍五入噪声），37 条**库里偏低**（典型是旧口径按 60kg 算、而该用户档案 75kg 的行），1 条偏高。所以「回填只降不升」在这批数据上不成立，要回填得双向写。
- **别用「时长被砍掉的比例」去推卡路里差额**（曾据此得出「65 条多报 15%」）：库里那行是否已含新时长，取决于 finish 发生在静止/车速管线上线之前还是之后，两者混在一起时这个推法一半是假阳性；而且它默认体重是 60kg，会把「档案 75kg」的行算成偏高而实际偏低。要测就直接按新口径重算比对。
- 回归：`test/calorie-weight.test.ts` CW1–CW5 逐条钉住 5 条路径（断言既不等于 60kg 结果、又精确等于档案体重结果），并钉住 finish 忽略客户端上报值（故意传 300kg）。

## 6. 安全与合规

| 项 | 方案 |
|---|---|
| 鉴权 | 所有业务接口 JWT 校验；`@fastify/jwt` + 路由级 `preHandler` |
| 数据隔离 | 所有查询强制带 `userId`（防越权）；对象 ID 校验 |
| OSS 安全 | STS 最小权限 + 目录隔离 `users/{userId}/` + 短时效（如 15 分钟） |
| 敏感字段 | openid 不出接口；日志脱敏 |
| HTTPS | 线上强制 HTTPS（Nginx 或云负载均衡终止 TLS） |
| 限流 | `@fastify/rate-limit`：登录/刷新限流，防爆破 |
| 隐私 | 轨迹数据属敏感个人信息：仅本人可查；删除接口同步清理 OSS 文件；提供账号注销（删除全部数据） |
| 微信侧 | 公众平台配置：合法域名（服务器域名 + 上传/下载域名 OSS）、申请"精确地理位置"等接口权限、隐私协议声明位置用途 |

```yaml
# docker-compose.yml
services:
  api:
    build: .
    ports: ["3000:3000"]
    environment:
      NODE_ENV: production

    restart: unless-stopped
```
